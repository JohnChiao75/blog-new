## 题目解析（[洛谷](https://www.luogu.com.cn/problem/AT_abc425_d) [AtCoder](https://atcoder.jp/contests/abc425/tasks/abc425_d)）

题目要求我们模拟一个 *“乌拉姆·沃伯顿自动机”*，即一个网格图，每个格子可以为白或黑色，仅当一个格子为白色且仅有一个相邻格子为黑色时将其染成白色，在稳定状态下（即网格不会在发生改变的状态）停止，输出黑色格子数量

:::info[样例解析]

## 解法

注意到在一个格子更新之后仅会影响周围四个格子，考虑 `bfs`

我们可以使用队列来存储更新的格子，**注意分层搜索**，`bfs` 的队列中最多会存在两层遍历的结果，在代码中，需要在结构中加入 `step` （代码中为 `tot`）来控制搜索的顺序，保证不会使不同次遍历的节点同时出队，见注释点 [1]

## Code（C++ [Submission #69683563)](https://atcoder.jp/contests/abc425/submissions/69683563)）

:::warning{open}
此代码包含用于测试的输出图的代码注释
:::

```cpp
#include <bits/stdc++.h>
using namespace std;
const int N = 300005;
int n, m, ans;
vector <char> g[N]; // 图，'#' 为黑色
vector <int> c[N]; // 遍历次数
int dx[5] = {0, 0, 1, 0, -1};
int dy[5] = {0, 1, 0, -1, 0};
struct Node
{
	int x, y, tot;
};
queue <Node> q;
int main()
{
	cin >> n >> m;
	for(int i = 1; i <= n; i++)
	{
		g[i].push_back(' ');
		c[i].push_back(0);
		for(int j = 1; j <= m; j++)
		{
			char t;
			cin >> t;
			g[i].push_back(t);
			if(t == '#')
			{
				q.push({i, j, 1}); // 已经染色的节点视为已遍历
				c[i].push_back(1); // 第一次遍历，即输入
				ans++;
			}
			else
			{
				c[i].push_back(0); // 未遍历
			}
		}
	}
	while(!q.empty())
	{
		Node cur = q.front();
		q.pop();
		for(int i = 1; i <= 4; i++)
		{
			int x = cur.x + dx[i];
			int y = cur.y + dy[i];
			if(x < 1 || y < 1 || x > n || y > m || g[x][y] == '#')
			{
				continue;
			}
			int cnt = 0;
			for(int j = 1; j <= 4; j++)
			{
				int tx = x + dx[j];
				int ty = y + dy[j];
				if(tx < 1 || ty < 1 || tx > n || ty > m)
				{
					continue;
				}
				if(g[tx][ty] == '#' && c[tx][ty] != cur.tot + 1 /*[1]: 此处是为了不使其他层的搜索结果干扰到这一层*/)
				{
					cnt++; // cnt: 相邻黑色格子
				}
			}
			if(cnt == 1)
			{
				ans++;
				g[x][y] = '#';
				c[x][y] = cur.tot + 1;
				q.push({x, y, cur.tot + 1});
				for(int j = 0; j < 4; j++)
				{
					int tx = x + dx[j];
					int ty = y + dy[j];
					if(tx < 1 || ty < 1 || tx > n || ty > m || g[tx][ty] == '#')
					{
						continue;
					}
					if(g[tx][ty] == '.')
					{
						q.push({tx, ty, cur.tot + 1}); // [1]: 保证将一层的节点搜完再进下一层
					}
				}
			}
		}
        // 测试代码
		cerr << "--------------------------------" << endl;
		for (int i = 1; i <= n; i++)
		{
			for (int j = 1; j <= m; j++)
			{
				cerr << g[i][j];
			}
			cerr << endl;
		}
	}
	cout << ans << endl;
	return 0;
}
```
